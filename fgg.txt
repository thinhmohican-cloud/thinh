import os
import asyncio
import threading
import tkinter as tk
from tkinter import messagebox, scrolledtext
import yt_dlp
import edge_tts
from moviepy.editor import VideoFileClip, AudioFileClip, concatenate_videoclips
import moviepy.video.fx.all as vfx

# --- CÁC HÀM XỬ LÝ GỐC ---
def download_clips(url, time_ranges, temp_folder):
    downloaded = []
    for i, (start, end) in enumerate(time_ranges):
        filename = f"{temp_folder}/clip_{i}.mp4"
        ydl_opts = {
            'format': 'bestvideo[ext=mp4]+bestaudio[ext=m4a]/best[ext=mp4]',
            'outtmpl': filename,
            'download_ranges': yt_dlp.utils.download_range_func(None, [(start, end)]),
            'force_keyframes_at_cuts': True, 'quiet': True
        }
        with yt_dlp.YoutubeDL(ydl_opts) as ydl:
            ydl.download([url])
        downloaded.append(filename)
    return downloaded

async def generate_ai_voice(text, output_audio):
    communicate = edge_tts.Communicate(text, "vi-VN-NamMinhNeural")
    await communicate.save(output_audio)

def render_final_video(video_files, audio_file, output_file):
    clips = [VideoFileClip(f).without_audio().fx(vfx.mirror_x).fx(vfx.speedx, 1.15) for f in video_files]
    final_video = concatenate_videoclips(clips, method="compose")
    ai_audio = AudioFileClip(audio_file)
    
    if final_video.duration > ai_audio.duration:
        final_video = final_video.subclip(0, ai_audio.duration)
    else:
        final_video = final_video.fx(vfx.loop, duration=ai_audio.duration)
        
    final_video = final_video.set_audio(ai_audio)
    final_video.write_videofile(output_file, fps=30, codec="libx264", audio_codec="aac", preset="ultrafast", threads=4)
    final_video.close()
    ai_audio.close()

# --- HÀM CHẠY TOOL KHI BẤM NÚT ---
def run_tool():
    url = url_entry.get()
    script = script_text.get("1.0", tk.END).strip()
    
    if not url or not script:
        messagebox.showerror("Lỗi", "Vui lòng nhập Link Video và Kịch bản!")
        return

    btn_start.config(state=tk.DISABLED, text="Đang xử lý... Vui lòng đợi!")
    
    def process():
        try:
            temp_folder = "./temp_files"
            os.makedirs(temp_folder, exist_ok=True)
            
            # Cấu hình cứng mốc thời gian minh họa (Giây 60-65 và 120-125)
            time_ranges = [(60, 65), (120, 125)] 
            audio_temp = f"{temp_folder}/voiceover.mp3"
            final_out = "Video_Review_Hoan_Chinh.mp4"
            
            videos = download_clips(url, time_ranges, temp_folder)
            asyncio.run(generate_ai_voice(script, audio_temp))
            if videos:
                render_final_video(videos, audio_temp, final_out)
                
            messagebox.showinfo("Thành công", f"Đã xuất video tại:\n{final_out}")
        except Exception as e:
            messagebox.showerror("Lỗi", f"Có lỗi xảy ra:\n{str(e)}")
        finally:
            btn_start.config(state=tk.NORMAL, text="TẠO VIDEO REVIEW")
            
    threading.Thread(target=process).start()


# --- VẼ GIAO DIỆN CỬA SỔ (ĐÃ SỬA LỖI HIỂN THỊ NÚT BẤM) ---
root = tk.Tk()
root.title("Tool Review YouTube Auto - Bán Content")
root.geometry("600x550") # Mở rộng cửa sổ

# GHIM NÚT BẤM XUỐNG DƯỚI CÙNG TRƯỚC TIÊN ĐỂ KHÔNG BAO GIỜ BỊ MẤT
btn_start = tk.Button(root, text="TẠO VIDEO REVIEW", font=("Arial", 14, "bold"), bg="green", fg="white", command=run_tool)
btn_start.pack(side=tk.BOTTOM, fill=tk.X, padx=20, pady=20)

# CÁC Ô NHẬP LIỆU BÊN TRÊN
tk.Label(root, text="LINK VIDEO YOUTUBE GỐC:", font=("Arial", 10, "bold")).pack(pady=(20, 5))
url_entry = tk.Entry(root, width=60, font=("Arial", 10))
url_entry.pack(pady=5)
url_entry.insert(0, "https://www.youtube.com/watch?v=0e3GPea1Tyg")

tk.Label(root, text="KỊCH BẢN REVIEW CỦA BẠN:", font=("Arial", 10, "bold")).pack(pady=5)

# Ô khung chữ tự động co giãn
frame_text = tk.Frame(root)
frame_text.pack(fill=tk.BOTH, expand=True, padx=20, pady=5)
script_text = scrolledtext.ScrolledText(frame_text, font=("Arial", 10))
script_text.pack(fill=tk.BOTH, expand=True)
script_text.insert(tk.END, "Chào các bạn, hôm nay chúng ta cùng xem video này nhé...")

root.mainloop()
